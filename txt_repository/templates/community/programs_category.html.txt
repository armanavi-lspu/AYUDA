{% block extra_js %}
<script>
document.addEventListener('DOMContentLoaded', function() {
    // Animate stat numbers
    document.querySelectorAll('.stat-bg .h2').forEach(stat => {
        const finalValue = parseInt(stat.textContent);
        if (isNaN(finalValue)) return;
        
        let currentValue = 0;
        const increment = Math.ceil(finalValue / 20);
        const timer = setInterval(() => {
            currentValue += increment;
            if (currentValue >= finalValue) {
                stat.textContent = finalValue;
                clearInterval(timer);
            } else {
                stat.textContent = currentValue;
            }
        }, 50);
    });

    // Add pulse animation to info button
    const infoBtn = document.querySelector('.info-icon-btn');
    if (infoBtn) {
        setInterval(() => {
            infoBtn.style.animation = 'pulse 0.5s ease';
            setTimeout(() => {
                infoBtn.style.animation = '';
            }, 500);
        }, 5000);
    }

    // Track modal open
    $('#howToApplyModal').on('show.bs.modal', function (e) {
        console.log('How to Apply modal opened');
    });
});

// Add CSS animation for pulse effect
const style = document.createElement('style');
style.textContent = `
    @keyframes pulse {
        0% { transform: scale(1); }
        50% { transform: scale(1.15); }
        100% { transform: scale(1); }
    }
`;
document.head.appendChild(style);
</script>
