<script>
    function markAsRead(notificationId) {
        fetch(`/community/notifications/${notificationId}/read`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' }
        })
        .then(response => response.json())
        .then(data => {
            if (data.success) {
                const card = document.querySelector(`[data-notification-id="${notificationId}"]`);
                if (card) {
                    card.classList.remove('is-unread');
                    const unreadDot = card.querySelector('.notification-unread-dot');
                    const readBtn = card.querySelector('.mark-read-btn');
                    if (unreadDot) unreadDot.remove();
                    if (readBtn) readBtn.remove();
                }

                updateUnreadCount(-1);
            }
        })
        .catch(error => console.error('Error:', error));
    }

    function markAllAsRead() {
        if (!confirm('Mark all notifications as read?')) return;
        
        fetch('/community/notifications/mark-all-read', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' }
        })
        .then(response => response.json())
        .then(data => {
            if (data.success) {
                document.querySelectorAll('.notification-feed-card.is-unread').forEach(card => {
                    card.classList.remove('is-unread');
                    const unreadDot = card.querySelector('.notification-unread-dot');
                    const readBtn = card.querySelector('.mark-read-btn');
                    if (unreadDot) unreadDot.remove();
                    if (readBtn) readBtn.remove();
                });

                const markAllReadBtn = document.getElementById('markAllReadBtn');
                if (markAllReadBtn) {
                    markAllReadBtn.remove();
                }
                updateUnreadCount(-9999);
            }
        })
        .catch(error => console.error('Error:', error));
    }

    function deleteNotification(notificationId) {
        if (!confirm('Delete this notification?')) return;
        
        fetch(`/community/notifications/${notificationId}/delete`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' }
        })
        .then(response => response.json())
        .then(data => {
            if (data.success) {
                const card = document.querySelector(`[data-notification-id="${notificationId}"]`);
                if (card) {
                    card.classList.add('fade-out');
                    setTimeout(() => card.remove(), 300);
                }
            }
        })
        .catch(error => console.error('Error:', error));
    }

    function updateUnreadCount(delta) {
        const badges = document.querySelectorAll('.js-unread-count');
        badges.forEach(badge => {
            const text = badge.textContent || '';
            const match = text.match(/\d+/);
            if (!match) return;

            let current = parseInt(match[0], 10);
            if (delta <= -9999) {
                current = 0;
            } else {
                current = Math.max(0, current + delta);
            }

            badge.innerHTML = '<i class="fas fa-star"></i> ' + current + ' Unread';
        });
    }

    function deleteAllReadNotifications() {
        if (!confirm('Delete all read notifications? This action cannot be undone.')) return;
        
        fetch('/community/notifications/delete-all-read', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' }
        })
        .then(response => response.json())
        .then(data => {
            if (data.success) {
                alert(`${data.count} notifications deleted`);
                location.reload();
            }
        })
        .catch(error => console.error('Error:', error));
    }
</script>
