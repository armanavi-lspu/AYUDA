{% block extra_js %}
<script>
    let currentImageIndex = 0;

    function getDownloadFilenameFromSrc(src) {
        if (!src) {
            return 'announcement-image';
        }

        const cleanSrc = src.split('?')[0];
        const segments = cleanSrc.split('/');
        return segments[segments.length - 1] || 'announcement-image';
    }

    function updateDownloadButton(src) {
        const downloadBtn = document.getElementById('downloadPreviewImage');
        if (!downloadBtn) {
            return;
        }

        downloadBtn.href = src || '#';
        downloadBtn.download = getDownloadFilenameFromSrc(src);
    }

    function openImagePreview(index) {
        if (announcementImages.length === 0) return;
        
        currentImageIndex = parseInt(index);
        const modal = document.getElementById('imagePreviewModal');
        const previewImage = document.getElementById('previewImage');
        const previewCaption = document.getElementById('previewCaption');
        const imageCounter = document.getElementById('imageCounter');
        
        previewImage.src = announcementImages[currentImageIndex].src;
        previewCaption.textContent = announcementImages[currentImageIndex].caption;
        updateDownloadButton(announcementImages[currentImageIndex].src);
        
        if (imageCounter) {
            imageCounter.textContent = currentImageIndex + 1;
        }
        
        modal.classList.add('show');
        document.body.style.overflow = 'hidden';
    }

    function closeImagePreview() {
        const modal = document.getElementById('imagePreviewModal');
        modal.classList.remove('show');
        document.body.style.overflow = 'auto';
    }

    function previousImage() {
        if (announcementImages.length === 0) return;
        currentImageIndex = (currentImageIndex - 1 + announcementImages.length) % announcementImages.length;
        updateImagePreview();
    }

    function nextImage() {
        if (announcementImages.length === 0) return;
        currentImageIndex = (currentImageIndex + 1) % announcementImages.length;
        updateImagePreview();
    }

    function updateImagePreview() {
        const previewImage = document.getElementById('previewImage');
        const previewCaption = document.getElementById('previewCaption');
        const imageCounter = document.getElementById('imageCounter');
        
        previewImage.src = announcementImages[currentImageIndex].src;
        previewCaption.textContent = announcementImages[currentImageIndex].caption;
        updateDownloadButton(announcementImages[currentImageIndex].src);
        
        if (imageCounter) {
            imageCounter.textContent = currentImageIndex + 1;
        }
    }

    // Close modal when clicking outside the image
    document.addEventListener('DOMContentLoaded', function() {
        const modal = document.getElementById('imagePreviewModal');
        modal.addEventListener('click', function(e) {
            if (e.target === modal) {
                closeImagePreview();
            }
        });
    });

    // Keyboard navigation
    document.addEventListener('keydown', function(e) {
        const modal = document.getElementById('imagePreviewModal');
        if (!modal.classList.contains('show')) return;
        
        if (e.key === 'Escape') {
            closeImagePreview();
        } else if (e.key === 'ArrowLeft') {
            if (announcementImages.length > 1) previousImage();
        } else if (e.key === 'ArrowRight') {
            if (announcementImages.length > 1) nextImage();
        }
    });

    // Initialize tooltips if using Bootstrap JS
    $(document).ready(function(){
        if (typeof $('[data-toggle="tooltip"]').tooltip === 'function') {
            $('[data-toggle="tooltip"]').tooltip();
        }
    });
</script>
