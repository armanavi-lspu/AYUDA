{% block extra_js %}
<script>
(function() {
    function toggleDateRangeInputs() {
        const datePreset = document.getElementById('date_preset');
        const customDateRangeRow = document.getElementById('customDateRangeRow');
        if (!datePreset || !customDateRangeRow) return;

        if (datePreset.value === 'custom') {
            customDateRangeRow.classList.remove('d-none');
        } else {
            customDateRangeRow.classList.add('d-none');
        }
    }

    function toggleFilterGroups() {
        const reportType = document.getElementById('report_type');
        const applicationFilters = document.getElementById('applicationFilters');
        const activityFilters = document.getElementById('activityFilters');
        if (!reportType) return;

        const selectedType = reportType.value;

        if (applicationFilters) {
            if (selectedType === 'applications' || selectedType === 'summary') {
                applicationFilters.classList.remove('d-none');
            } else {
                applicationFilters.classList.add('d-none');
            }
        }

        if (activityFilters) {
            if (selectedType === 'activity_logs') {
                activityFilters.classList.remove('d-none');
            } else {
                activityFilters.classList.add('d-none');
            }
        }
    }

    document.addEventListener('DOMContentLoaded', function() {
        const datePreset = document.getElementById('date_preset');
        const reportType = document.getElementById('report_type');

        toggleDateRangeInputs();
        toggleFilterGroups();

        if (datePreset) {
            datePreset.addEventListener('change', toggleDateRangeInputs);
        }

        if (reportType) {
            reportType.addEventListener('change', toggleFilterGroups);
        }
    });
})();
</script>
{% endblock %}
