{% block extra_js %}
<script>
$(document).ready(function() {
    let selectedUserId = null;
    let searchTimeout = null;
    let pendingRemoval = null;

    function getToastStackContainer() {
        let container = document.getElementById('ayudaToastStackContainer');
        if (!container) {
            container = document.createElement('div');
            container.id = 'ayudaToastStackContainer';
            container.style.cssText = 'position: fixed; top: 80px; right: 20px; z-index: 9999; display: flex; flex-direction: column; gap: 10px; align-items: flex-end; width: min(92vw, 400px); pointer-events: none;';
            document.body.appendChild(container);
        }
        return container;
    }
    
    // Helper functions for showing messages
    function showSuccessMessage(message) {
        const alertHtml = `
            <div class="alert alert-success alert-dismissible fade show mb-0 toast-animate-in" style="width: 400px; max-width: calc(100vw - 1rem); box-shadow: 0 8px 20px rgba(0,0,0,0.18); pointer-events: auto;">
                <i class="fas fa-check-circle mr-2"></i>
                ${message}
                <button type="button" class="close" data-dismiss="alert">
                    <span>&times;</span>
                </button>
            </div>
        `;
        const $toast = $(alertHtml);
        $(getToastStackContainer()).append($toast);
        
        // Auto dismiss after 5 seconds
        setTimeout(function() {
            $toast[0].classList.remove('toast-animate-in');
            $toast[0].classList.add('toast-animate-out');
            setTimeout(() => {
                $toast.alert('close');
            }, 300);
        }, 5000);
    }
    
    function showErrorMessage(message) {
        const alertHtml = `
            <div class="alert alert-danger alert-dismissible fade show mb-0 toast-animate-in" style="width: 400px; max-width: calc(100vw - 1rem); box-shadow: 0 8px 20px rgba(0,0,0,0.18); pointer-events: auto;">
                <i class="fas fa-exclamation-triangle mr-2"></i>
                ${message}
                <button type="button" class="close" data-dismiss="alert">
                    <span>&times;</span>
                </button>
            </div>
        `;
        const $toast = $(alertHtml);
        $(getToastStackContainer()).append($toast);
        
        // Auto dismiss after 5 seconds
        setTimeout(function() {
            $toast[0].classList.remove('toast-animate-in');
            $toast[0].classList.add('toast-animate-out');
            setTimeout(() => {
                $toast.alert('close');
            }, 300);
        }, 5000);
    }

    function resetRemoveFlow() {
        pendingRemoval = null;
        $('#removeCommunityUserId').val('');
        $('#deleteKeywordInput').val('');
        $('#confirmRemoveBeneficiaryBtn').prop('disabled', true).html('<i class="fas fa-user-times mr-1"></i>Remove Beneficiary');
    }

    $(document).on('click', '.open-remove-beneficiary-modal', function() {
        pendingRemoval = {
            communityUserId: $(this).data('community-user-id'),
            userName: $(this).data('user-name') || 'this beneficiary',
            categoryName: $(this).data('category-name') || '{{ category_name }}'
        };

        $('#removeBeneficiaryName').text(pendingRemoval.userName);
        $('#removeBeneficiaryCategory').text(pendingRemoval.categoryName);
        $('#removeBeneficiaryConfirmModal').modal('show');
    });

    $('#proceedToDeleteTypeBtn').on('click', function() {
        $('#removeBeneficiaryConfirmModal').modal('hide');
        setTimeout(function() {
            $('#removeBeneficiaryTypeModal').modal('show');
        }, 220);
    });

    $('#deleteKeywordInput').on('input', function() {
        const canDelete = $(this).val().trim() === 'Delete';
        $('#confirmRemoveBeneficiaryBtn').prop('disabled', !canDelete);
    });

    $('#confirmRemoveBeneficiaryBtn').on('click', function() {
        if (!pendingRemoval || !pendingRemoval.communityUserId) {
            return;
        }

        const $btn = $(this);
        $btn.prop('disabled', true).html('<i class="fas fa-spinner fa-spin mr-1"></i>Removing...');

        $('#removeCommunityUserId').val(pendingRemoval.communityUserId);
        $('#removeReturnUrl').val(window.location.pathname + window.location.search);

        $('#removeBeneficiaryForm').trigger('submit');
    });

    $('#removeBeneficiaryConfirmModal').on('hidden.bs.modal', function() {
        // Keep state only if moving to step 2.
        if (!$('#removeBeneficiaryTypeModal').hasClass('show')) {
            $('#deleteKeywordInput').val('');
            $('#confirmRemoveBeneficiaryBtn').prop('disabled', true).html('<i class="fas fa-user-times mr-1"></i>Remove Beneficiary');
        }
    });

    $('#removeBeneficiaryTypeModal').on('hidden.bs.modal', function() {
        resetRemoveFlow();
    });
    
    // Search functionality
    $('#userSearch').on('input', function() {
        const query = $(this).val().trim();
        
        // Clear previous timeout
        if (searchTimeout) {
            clearTimeout(searchTimeout);
        }
        
        // Clear selected user
        selectedUserId = null;
        $('#selectedUserInfo').hide();
        $('#notifyUserBtn').prop('disabled', true);
        
        if (query.length < 2) {
            $('#searchResults').html(`
                <div class="text-center text-muted py-4">
                    <i class="fas fa-users fa-2x mb-2"></i>
                    <p>Start typing to search for eligible users</p>
                </div>
            `);
            return;
        }
        
        // Show loading
        $('#searchLoading').show();
        $('#searchResults').empty();
        
        // Debounce search
        searchTimeout = setTimeout(function() {
            searchEligibleUsers(query);
        }, 300);
    });
    
    function searchEligibleUsers(query) {
        $.ajax({
            url: '{{ url_for("admin.search_eligible_users") }}',
            method: 'GET',
            data: {
                query: query,
                category: '{{ category }}'
            },
            success: function(data) {
                $('#searchLoading').hide();
                displaySearchResults(data.users);
            },
            error: function() {
                $('#searchLoading').hide();
                $('#searchResults').html(`
                    <div class="alert alert-danger">
                        <i class="fas fa-exclamation-triangle mr-2"></i>
                        Error searching for users. Please try again.
                    </div>
                `);
            }
        });
    }
    
    function displaySearchResults(users) {
        if (users.length === 0) {
            $('#searchResults').html(`
                <div class="text-center text-muted py-4">
                    <i class="fas fa-user-slash fa-2x mb-2"></i>
                    <p>No eligible users found matching your search.</p>
                </div>
            `);
            return;
        }
        
        let html = '<div class="list-group">';
        users.forEach(function(user) {
            const verificationStatus = user.verification_status || 'pending';
            const verificationColor = verificationStatus === 'approved' ? 'success' : 
                                    verificationStatus === 'pending' ? 'warning' : 'danger';
            
            html += `
                <div class="list-group-item list-group-item-action user-item" data-user-id="${user.id}">
                    <div class="d-flex justify-content-between align-items-start">
                        <div class="flex-grow-1">
                            <h6 class="mb-1">${user.name}</h6>
                            <p class="mb-1 text-muted">${user.email}</p>
                            <small class="text-muted">
                                <i class="fas fa-map-marker-alt mr-1"></i>
                                ${user.barangay || 'N/A'}, ${user.municipality || 'Mabitac'}
                            </small>
                        </div>
                        <div class="text-right">
                            <span class="badge badge-${verificationColor}">
                                ${verificationStatus.charAt(0).toUpperCase() + verificationStatus.slice(1)}
                            </span>
                            ${user.age ? `<br><small class="text-muted">Age: ${user.age}</small>` : ''}
                        </div>
                    </div>
                </div>
            `;
        });
        html += '</div>';
        
        $('#searchResults').html(html);
        
        // Add click handlers
        $('.user-item').on('click', function() {
            const userId = $(this).data('user-id');
            const user = users.find(u => u.id === userId);
            selectUser(user);
        });
    }
    
    function selectUser(user) {
        selectedUserId = user.id;
        
        // Update selected user display
        const verificationStatus = user.verification_status || 'pending';
        const verificationColor = verificationStatus === 'approved' ? 'success' : 
                                verificationStatus === 'pending' ? 'warning' : 'danger';
        
        $('#selectedUserDetails').html(`
            <div class="row">
                <div class="col-md-6">
                    <strong>${user.name}</strong><br>
                    <span class="text-muted">${user.email}</span><br>
                    <small><i class="fas fa-map-marker-alt mr-1"></i>${user.barangay || 'N/A'}, ${user.municipality || 'Mabitac'}</small>
                </div>
                <div class="col-md-6 text-md-right">
                    <span class="badge badge-${verificationColor}">
                        Verification: ${verificationStatus.charAt(0).toUpperCase() + verificationStatus.slice(1)}
                    </span>
                    ${user.age ? `<br><small class="text-muted">Age: ${user.age}</small>` : ''}
                </div>
            </div>
        `);
        
        $('#selectedUserInfo').show();
        $('#notifyUserBtn').prop('disabled', false);
        
        // Highlight selected user
        $('.user-item').removeClass('active');
        $(`.user-item[data-user-id="${selectedUserId}"]`).addClass('active');
    }
    
    // Notify user button
    $('#notifyUserBtn').on('click', function() {
        if (!selectedUserId) return;
        
        const button = $(this);
        button.prop('disabled', true).html('<i class="fas fa-spinner fa-spin mr-1"></i>Notifying...');
        
        $.ajax({
            url: '{{ url_for("admin.notify_subsidy_eligibility") }}',
            method: 'POST',
            data: {
                user_id: selectedUserId,
                category: '{{ category }}',
                category_name: '{{ category_name }}'
            },
            success: function(data) {
                if (data.success) {
                    // Show success message with flash-style alert
                    showSuccessMessage(data.message);
                    
                    // Close modal and refresh page
                    $('#addEligibleUserModal').modal('hide');
                    setTimeout(function() {
                        location.reload();
                    }, 1500);
                } else {
                    showErrorMessage(data.message || 'Failed to notify user');
                    button.prop('disabled', false).html('<i class="fas fa-bell mr-1"></i>Notify User About Eligibility');
                }
            },
            error: function() {
                showErrorMessage('Error occurred while notifying user');
                button.prop('disabled', false).html('<i class="fas fa-bell mr-1"></i>Notify User About Eligibility');
            }
        });
    });
    
    // Reset modal when closed
    $('#addEligibleUserModal').on('hidden.bs.modal', function() {
        $('#userSearch').val('');
        selectedUserId = null;
        $('#selectedUserInfo').hide();
        $('#notifyUserBtn').prop('disabled', true);
        $('#searchResults').html(`
            <div class="text-center text-muted py-4">
                <i class="fas fa-users fa-2x mb-2"></i>
                <p>Start typing to search for eligible users</p>
            </div>
        `);
    });
});
