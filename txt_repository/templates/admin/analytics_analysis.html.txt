let analyticsRealtimeListenerBound = false;
let analyticsRefreshInProgress = false;
let analyticsPendingUpdateMessage = '';

function refreshAnalyticsAnalysisPage(message) {
    if (analyticsRefreshInProgress) {
        return;
    }

    const currentRoot = document.getElementById('adminAnalyticsRoot');
    if (!currentRoot) {
        window.location.reload();
        return;
    }

    analyticsRefreshInProgress = true;

    fetch(`${window.location.pathname}${window.location.search}`, {
        headers: {
            'X-Requested-With': 'XMLHttpRequest'
        },
        credentials: 'same-origin'
    })
        .then(response => {
            if (!response.ok) {
                throw new Error(`Failed to refresh analytics (${response.status})`);
            }
            return response.text();
        })
        .then(html => {
            const parser = new DOMParser();
            const doc = parser.parseFromString(html, 'text/html');
            const nextRoot = doc.getElementById('adminAnalyticsRoot');

            if (!nextRoot) {
                throw new Error('Updated analytics content not found');
            }

            currentRoot.replaceWith(nextRoot);
            initializeAnalyticsAnalysisPage();

            if (typeof showRealtimeToast === 'function') {
                showRealtimeToast(message || 'Analytics refreshed from live updates.');
            }
        })
        .catch(error => {
            console.error('Soft refresh failed, falling back to full reload:', error);
            window.location.reload();
        })
        .finally(() => {
            analyticsRefreshInProgress = false;
        });
}

function initializeAnalyticsAnalysisPage() {
    console.log('📊 Analytics page loaded');

    function showAnalyticsUpdateNotice(message) {
        const notice = document.getElementById('analyticsUpdateNotice');
        const messageNode = document.getElementById('analyticsUpdateNoticeMessage');
        if (!notice || !messageNode) {
            return;
        }

        if (message) {
            analyticsPendingUpdateMessage = message;
        }

        messageNode.textContent = analyticsPendingUpdateMessage || 'A new analytics update is available.';
        notice.classList.remove('d-none');
    }

    function hideAnalyticsUpdateNotice() {
        const notice = document.getElementById('analyticsUpdateNotice');
        if (!notice) {
            return;
        }

        notice.classList.add('d-none');
        analyticsPendingUpdateMessage = '';
    }

    const refreshGraphsBtn = document.getElementById('refreshAnalyticsGraphsBtn');
    if (refreshGraphsBtn && !refreshGraphsBtn.dataset.bound) {
        refreshGraphsBtn.dataset.bound = '1';
        refreshGraphsBtn.addEventListener('click', function() {
            hideAnalyticsUpdateNotice();
            refreshAnalyticsAnalysisPage('Analytics graphs refreshed with latest updates.');
        });
    }
    
    /**
     * Helper function to build forecast data array for Chart.js
     * Creates an array with null values for historical period, 
     * then connects from lastValue to forecast values
     * @param {number} historicalLength - Number of historical data points
     * @param {number} lastValue - Last value from historical data (connection point)
     * @param {Array} forecastData - Array of forecast values
     * @returns {Array} - Data array for chart with nulls for historical and forecast values
     */
    function buildForecastDataArray(historicalLength, lastValue, forecastData) {
        const nullsForHistorical = Array(historicalLength).fill(null);
        return [...nullsForHistorical, lastValue, ...forecastData];
    }
    
    /**
     * Remove loading spinner and show chart with animation
     */
    function showChart(containerId) {
        const container = document.getElementById(containerId);
        if (container) {
            const loader = container.querySelector('.chart-loading');
            if (loader) {
                loader.style.opacity = '0';
                setTimeout(() => loader.remove(), 300);
            }
        }
    }

    function clearForecastStatus(chartContainer) {
        if (!chartContainer) return;
        const existing = chartContainer.querySelector('.forecast-status-message');
        if (existing) {
            existing.remove();
        }
    }

    function renderForecastStatus(chartContainer, message) {
        if (!chartContainer) return;
        clearForecastStatus(chartContainer);

        const note = document.createElement('div');
        note.className = 'forecast-status-message alert alert-warning mt-3 mb-0';
        note.textContent = message;
        chartContainer.appendChild(note);
    }

        console.log('   Completion rate element:', completionRateBarangayDataElement ? '✅ Found' : '❌ Missing');
        console.log('   Barangay element:', barangayDataElement ? '✅ Found' : '❌ Missing');
        console.log('   Municipality chart removed from this page');
        
        if (!timeDataElement || !typeDataElement || !completionRateBarangayDataElement || !barangayDataElement) {
            console.error('❌ ERROR: Required data elements not found in DOM!');

            console.error('   Check that analytics_analysis route is passing all JSON variables:');
            console.error('   - applicants_over_time_json');
            console.error('   - applications_by_type_json');
            console.error('   - completion_rate_by_barangay_json');
            console.error('   - applicants_by_barangay_json');
            alert('Error: Chart data not found. Please check server logs.');
            return;
        }
        
        console.log('📦 Parsing JSON data...');
        applicantsOverTimeData = JSON.parse(timeDataElement.textContent);
        applicationsByTypeData = JSON.parse(typeDataElement.textContent);
        completionRateByBarangayData = JSON.parse(completionRateBarangayDataElement.textContent);

        console.log('Applications By Type:', applicationsByTypeData);

        console.log('Applicants By Barangay:', applicantsByBarangayData);
        
        // Show debug panel with data

    // 1. Applicants Time Series Chart

    // Helper: parse labels like 'Jan-2023' into Date objects (month start)
    function parseMonthLabelToDate(label) {
        try {
            const cleaned = String(label || '').trim().replace(/\s+/g, '-');
            const parts = cleaned.split('-');
            if (parts.length < 2) {
                return new Date();
            }
            const monthName = parts[0].toLowerCase().slice(0,3);
            const year = parseInt(parts[1], 10) || new Date().getFullYear();
            const map = { jan:0, feb:1, mar:2, apr:3, may:4, jun:5, jul:6, aug:7, sep:8, oct:9, nov:10, dec:11 };
            const m = map[monthName] !== undefined ? map[monthName] : 0;
            return new Date(year, m, 1);
        } catch (e) {
            return new Date();
        }
    }

    // Helper: simple linear regression (x=0..n-1) -> {slope, intercept}
    function computeLinearTrend(values) {
        const n = (values || []).length;
        if (n === 0) return { slope: 0, intercept: 0 };
        const meanX = (n - 1) / 2;
        const meanY = values.reduce((a, b) => a + (Number(b) || 0), 0) / n;
        let num = 0, den = 0;
        for (let i = 0; i < n; i++) {
            const x = i;
            num += (x - meanX) * ((Number(values[i]) || 0) - meanY);
            den += (x - meanX) * (x - meanX);
        }
        const slope = den === 0 ? 0 : num / den;
        const intercept = meanY - slope * meanX;
        return { slope, intercept };
    }

    // Client-side forecast for the MSWD Mabitac static series (12 months)
    function loadStaticMabitacForecast(staticSeries) {
        if (!forecastApplicantsCtx) return;
        const chartContainer = forecastApplicantsCtx.parentElement;
        chartContainer.style.opacity = '0.5';
        clearForecastStatus(chartContainer);

        const historicalLabels = Array.isArray(staticSeries.labels) ? staticSeries.labels.slice() : [];
        const historicalData = Array.isArray(staticSeries.data) ? staticSeries.data.map(v => Math.max(0, Math.round(Number(v) || 0))) : [];
        if (historicalLabels.length === 0 || historicalLabels.length !== historicalData.length) {
            renderForecastStatus(chartContainer, 'Static Mabitac series is invalid or missing.');
            chartContainer.style.opacity = '1';
            showChart('forecastApplicantsContainer');
            return;
        }

        // Trend and seasonal decomposition (simple)
        const n = historicalData.length;
        const lastDate = parseMonthLabelToDate(historicalLabels[historicalLabels.length - 1]);
        const { slope, intercept } = computeLinearTrend(historicalData);

        // Seasonal offsets: average residual per calendar month
        const seasonOffsets = new Array(12).fill(0);
        const seasonCounts = new Array(12).fill(0);
        for (let i = 0; i < n; i++) {
            const dt = parseMonthLabelToDate(historicalLabels[i]);
            const m = dt.getMonth();
            const trendVal = intercept + slope * i;
            const residual = historicalData[i] - trendVal;
            seasonOffsets[m] += residual;
            seasonCounts[m] += 1;
        }
        for (let m = 0; m < 12; m++) {
            if (seasonCounts[m] > 0) seasonOffsets[m] = seasonOffsets[m] / seasonCounts[m];
            else seasonOffsets[m] = 0;
        }

        // Forecast next 12 months
        const forecastPeriods = 12;
        const forecastValues = [];
        const forecastLabels = [];
        for (let t = 1; t <= forecastPeriods; t++) {
            const xFuture = n + (t - 1);
            const futureMonth = (lastDate.getMonth() + t) % 12;
            const seasonal = seasonOffsets[futureMonth] || 0;
            let value = intercept + slope * xFuture + seasonal;
            value = Math.max(0, Math.round(value));
            forecastValues.push(value);
            const futureDate = new Date(lastDate.getFullYear(), lastDate.getMonth() + t, 1);
            forecastLabels.push(futureDate.toLocaleDateString('en-US', { month: 'short', year: 'numeric', timeZone: 'Asia/Manila' }));
        }

        // Store last data for toggling
        lastForecastData = {
            historicalLabels: historicalLabels,
            historicalData: historicalData,
            forecastLabels: forecastLabels,
            forecastData: forecastValues,
            source: 'static'
        };

        // Render chart
        if (applicantsForecastChart) applicantsForecastChart.destroy();
        const lastValue = historicalData.length > 0 ? historicalData[historicalData.length - 1] : 0;

        applicantsForecastChart = new Chart(forecastApplicantsCtx.getContext('2d'), {
            type: 'line',
            data: {
                labels: [...historicalLabels, ...forecastLabels],
                datasets: [
                    {
                        label: 'Historical Data',
                        data: [...historicalData, ...Array(forecastValues.length).fill(null)],
                        borderColor: 'rgb(78, 115, 223)',
                        backgroundColor: 'rgba(78, 115, 223, 0.1)',
                        fill: true,
                        tension: 0.4,
                        borderWidth: 3,
                        pointBackgroundColor: 'rgb(78, 115, 223)',
                        pointBorderColor: '#fff',
                        pointBorderWidth: 2,
                        pointRadius: 4
                    },
                    {
                        label: 'Forecast (12 months) - Static Mabitac',
                        data: buildForecastDataArray(historicalData.length, lastValue, forecastValues),
                        borderColor: 'rgb(246, 194, 62)',
                        backgroundColor: 'rgba(246, 194, 62, 0.1)',
                        fill: true,
                        tension: 0.4,
                        borderWidth: 3,
                        borderDash: [5, 5],
                        pointBackgroundColor: 'rgb(246, 194, 62)',
                        pointBorderColor: '#fff',
                        pointBorderWidth: 2,
                        pointRadius: 4
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        display: true,
                        position: 'top'
                    },
                    tooltip: {
                        mode: 'index',
                        intersect: false,
                        callbacks: {
                            label: function(context) {
                                let label = context.dataset.label || '';
                                if (label) label += ': ';
                                if (context.parsed.y !== null) label += Math.round(context.parsed.y);
                                return label;
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        ticks: { stepSize: 1 }
                    }
                }
            }
        });

        chartContainer.style.opacity = '1';
        showChart('forecastApplicantsContainer');
    }

    // Render grouped-by-year view from lastForecastData
    function renderGroupedYearFromLastData() {
        if (!lastForecastData || !lastForecastData.historicalLabels) {
            renderForecastStatus(forecastApplicantsCtx.parentElement, 'No historical series available to group by year.');
            return;
        }

        // Build a map of year -> array of 12 months (fill null if missing)
        const monthNames = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
        const yearMap = {};
        const labels = lastForecastData.historicalLabels;
        const values = lastForecastData.historicalData;
        for (let i = 0; i < labels.length; i++) {
            const lbl = labels[i];
            const dt = parseMonthLabelToDate(lbl);
            const y = dt.getFullYear();
            const m = dt.getMonth();
            yearMap[y] = yearMap[y] || new Array(12).fill(null);
            yearMap[y][m] = Number(values[i]) || 0;
        }

        // Use only years 2023,2024,2025 if present, in that order (per request)
        const yearsToShow = [2023, 2024, 2025].filter(y => yearMap[y]);
        if (yearsToShow.length === 0) {
            renderForecastStatus(forecastApplicantsCtx.parentElement, 'No yearly groups found in the historical data.');
            return;
        }

        const palette = ['rgb(78, 115, 223)','rgb(246, 194, 62)','rgb(134,134,134)'];
        if (applicantsForecastChart) applicantsForecastChart.destroy();

        // We'll render months twice: historical (Jan..Dec) then forecast (Jan..Dec - Forecast)
        const historicalLabelsPart = monthNames;
        const forecastLabelsPart = monthNames.map(m => `${m} (Forecast)`);
        const allLabels = [...historicalLabelsPart, ...forecastLabelsPart];

        const datasets = yearsToShow.map((y, idx) => ({
            label: String(y),
            // historical values for first 12 slots, then nulls for forecast slots
            data: yearMap[y].map(v => v === null ? null : Math.max(0, Math.round(v))).concat(new Array(12).fill(null)),
            borderColor: palette[idx % palette.length],
            backgroundColor: 'transparent',
            borderWidth: 3,
            tension: 0.4,
            pointRadius: 4
        }));

        // Add a single-line series showing the 12-month forecast values in the
        // second block of labels so it appears beside historical months.
        if (lastForecastData && Array.isArray(lastForecastData.forecastLabels) && lastForecastData.forecastLabels.length > 0) {
            const forecastByMonth = new Array(12).fill(null);
            try {
                for (let i = 0; i < lastForecastData.forecastLabels.length; i++) {
                    const fl = lastForecastData.forecastLabels[i];
                    const fv = Number((lastForecastData.forecastData && lastForecastData.forecastData[i]) || null);
                    const dt = parseMonthLabelToDate(fl);
                    const m = dt.getMonth();
                    forecastByMonth[m] = isNaN(fv) ? null : Math.max(0, Math.round(fv));
                }

                // Place forecast values in the second block of 12 slots
                const forecastSeries = new Array(12).fill(null).concat(forecastByMonth);
                datasets.push({
                    label: '12-month Forecast',
                    data: forecastSeries,
                    borderColor: 'rgb(231, 74, 59)',
                    backgroundColor: 'transparent',
                    borderWidth: 3,
                    tension: 0.4,
                    pointRadius: 4,
                    borderDash: [6,4]
                });
            } catch (e) {
                console.error('Error building grouped forecast series:', e);
            }
        }

        applicantsForecastChart = new Chart(forecastApplicantsCtx.getContext('2d'), {
            type: 'line',
            data: {
                labels: allLabels,
                datasets: datasets
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { display: true, position: 'top' },
                    tooltip: { mode: 'index', intersect: false }
                },
                scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } }
            }
        });

        showChart('forecastApplicantsContainer');
    }

            loadArimaForecast();
            return;

        clearForecastStatus(chartContainer);

        const forecastLabels = lastForecastData.forecastLabels || [];
        const forecastData = lastForecastData.forecastData || [];
        const confidenceLower = lastForecastData.confidenceLower || [];
        const confidenceUpper = lastForecastData.confidenceUpper || [];

        if (applicantsForecastChart) applicantsForecastChart.destroy();

        const datasets = [
            {
                label: 'Historical Data',
                data: [...historicalData, ...Array(forecastLabels.length).fill(null)],
                borderColor: 'rgb(78, 115, 223)',
                backgroundColor: 'rgba(78, 115, 223, 0.1)',
                fill: true,
                tension: 0.4,
                borderWidth: 3
            }
        ];

        if (forecastData.length > 0) {
            const lastValue = historicalData.length > 0 ? historicalData[historicalData.length - 1] : 0;
            datasets.push({
                label: 'Forecast',
                data: buildForecastDataArray(historicalData.length, lastValue, forecastData),
                borderColor: 'rgb(246, 194, 62)',
                backgroundColor: 'rgba(246, 194, 62, 0.1)',
                fill: true,
                tension: 0.4,
                borderWidth: 3,
                borderDash: [5,5]
            });

                datasets.push({ label: 'CI Lower', data: [...Array(historicalData.length).fill(null), ...confidenceLower], borderColor: 'rgba(246,194,62,0.3)', backgroundColor: 'transparent', fill: false, pointRadius:0, borderWidth:1, hidden: true });
            }
        }

        applicantsForecastChart = new Chart(forecastApplicantsCtx.getContext('2d'), {
            type: 'line',
            data: { labels: [...historicalLabels, ...forecastLabels], datasets: datasets },
            options: {
                responsive: true,
