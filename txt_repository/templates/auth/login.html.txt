<script>
document.addEventListener('DOMContentLoaded', function() {
    const loginForm = document.getElementById('loginForm');
    const loginBtn = document.getElementById('loginBtn');
    const loginAlert = document.getElementById('loginAlert');
    const emailInput = document.getElementById('email');
    const passwordInput = document.getElementById('password');
    const emailError = document.getElementById('emailError');
    const passwordError = document.getElementById('passwordError');

    const togglePassword = document.getElementById('togglePassword');
    const password = document.getElementById('password');
    const eyeIcon = document.getElementById('eyeIcon');

    if (togglePassword && password && eyeIcon) {
        togglePassword.addEventListener('click', function() {
            const type = password.getAttribute('type') === 'password' ? 'text' : 'password';
            password.setAttribute('type', type);

            if (type === 'password') {
                eyeIcon.classList.remove('fa-eye-slash');
                eyeIcon.classList.add('fa-eye');
            } else {
                eyeIcon.classList.remove('fa-eye');
                eyeIcon.classList.add('fa-eye-slash');
            }
        });
    }

    function showAlert(message) {
        loginAlert.textContent = message;
        loginAlert.classList.remove('d-none');
    }

    function clearAlert() {
        loginAlert.classList.add('d-none');
        loginAlert.textContent = '';
    }

    function setFieldError(field, errorEl, message) {
        field.classList.add('is-invalid');
        if (message) {
            errorEl.textContent = message;
        }
        errorEl.classList.add('show');
    }

    function clearFieldError(field, errorEl) {
        field.classList.remove('is-invalid');
        errorEl.classList.remove('show');
    }

    function validateInputs() {
        let isValid = true;
        const email = emailInput.value.trim();
        const passwordValue = passwordInput.value;
        const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

        clearFieldError(emailInput, emailError);
        clearFieldError(passwordInput, passwordError);
        clearAlert();

        if (!email) {
            setFieldError(emailInput, emailError, 'Email is required.');
            isValid = false;
        } else if (!emailPattern.test(email)) {
            setFieldError(emailInput, emailError, 'Please enter a valid email address.');
            isValid = false;
        }

        if (!passwordValue) {
            setFieldError(passwordInput, passwordError, 'Password is required.');
            isValid = false;
        }

        return isValid;
    }

    if (loginForm) {
        loginForm.addEventListener('submit', async function(e) {
            e.preventDefault();

            if (!validateInputs()) {
                return;
            }

            const formData = new FormData(loginForm);
            const originalBtnText = loginBtn.textContent;
            loginBtn.disabled = true;
            loginBtn.textContent = 'Signing in...';

            try {
                const csrfTokenInput = loginForm.querySelector('input[name="csrf_token"]');
                const csrfToken = csrfTokenInput ? csrfTokenInput.value : '';
                const response = await fetch(loginForm.action || window.location.pathname, {
                    method: 'POST',
                    headers: {
                        'X-Requested-With': 'XMLHttpRequest',
                        'X-CSRFToken': csrfToken
                    },
                    body: formData
                });

                const payload = await response.json();

                if (response.ok && payload.success && payload.redirect_url) {
                    window.location.href = payload.redirect_url;
                    return;
                }

                showAlert(payload.message || 'Login failed. Please try again.');

                if (payload.field === 'email') {
                    setFieldError(emailInput, emailError, payload.message);
                } else if (payload.field === 'password') {
                    setFieldError(passwordInput, passwordError, payload.message);
                }
            } catch (error) {
                showAlert('Unable to process login right now. Please try again.');
            } finally {
                loginBtn.disabled = false;
                loginBtn.textContent = originalBtnText;
            }
        });
    }

    emailInput.addEventListener('input', function() {
        clearFieldError(emailInput, emailError);
        clearAlert();
    });

    passwordInput.addEventListener('input', function() {
        clearFieldError(passwordInput, passwordError);
        clearAlert();
    });
});
</script>
