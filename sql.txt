CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    middle_name VARCHAR(50),
    last_name VARCHAR(50) NOT NULL,     
    role VARCHAR(20) NOT NULL,
    profile_pic VARCHAR(255),
    last_activity TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create indexes in the users table for easier accessibility
CREATE INDEX idx_user_email ON users(email);
CREATE INDEX idx_user_role ON users(role);
CREATE INDEX idx_user_created_at ON users(created_at);

-- Create table for financial programs 
CREATE TABLE programs (
    id SERIAL PRIMARY KEY,
    program_name VARCHAR(200) NOT NULL,
    program_type VARCHAR(50) NOT NULL,
    program_period VARCHAR(50) NOT NULL,
    date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--Create index on programs 
CREATE INDEX idx_programs_user_id ON programs(user_id);
CREATE INDEX idx_programs_type ON programs(program_type);
CREATE INDEX idx_programs_period ON programs(program_period);
CREATE INDEX idx_programs_date ON programs(date DESC);
