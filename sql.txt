-- Create table for financial programs 
CREATE TABLE programs (
    id SERIAL PRIMARY KEY,
    program_name VARCHAR(200) NOT NULL,
    program_type VARCHAR(50) NOT NULL,
    program_period VARCHAR(50) NOT NULL,
    priority_group VARCHAR(255), -- Target beneficiaries
    beneficiary_limit INTEGER, -- Maximum number of beneficiaries
    income_range VARCHAR(100), -- Target income level
    start_date DATE, -- Program start date
    end_date DATE, -- Program end date / deadline
    description TEXT,
    date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    user_id INTEGER REFERENCES users(id) ON DELETE SET NULL, -- admin who created the program
    file_attachment_id INTEGER REFERENCES file_attachment(id) ON DELETE SET NULL, -- application form file
    is_active BOOLEAN DEFAULT TRUE, -- Track active programs
    allow_online_upload BOOLEAN DEFAULT TRUE, -- Enable/disable online document submission
    enable_application_slip BOOLEAN DEFAULT TRUE -- Enable/disable application slip generation
);

CREATE INDEX idx_programs_is_active ON programs(is_active);

-- Create table for applications
CREATE TABLE applications (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    program_id INTEGER NOT NULL REFERENCES programs(id) ON DELETE CASCADE,
    application_status VARCHAR(20) NOT NULL DEFAULT 'pending', -- 'pending', 'approved', 'rejected', 'active', 'completed'
    document_upload_status VARCHAR(20) DEFAULT 'pending',
    application_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    review_date TIMESTAMP,
    reviewed_by INTEGER REFERENCES users(id) ON DELETE SET NULL,
    submission_deadline TIMESTAMP, -- Deadline for document submission
    
    -- Claim scheduling fields
    claim_date TIMESTAMP, -- Scheduled date for claiming assistance
    claim_time VARCHAR(10), -- Time slot
    claim_location VARCHAR(255), -- Location for claiming
    claim_instructions TEXT, -- Special instructions for claiming
    claim_status VARCHAR(20) DEFAULT 'not_scheduled', -- not_scheduled, scheduled, claimed, missed
    claim_scheduled_by INTEGER REFERENCES users(id) ON DELETE SET NULL, -- Admin who scheduled
    claim_scheduled_at TIMESTAMP, -- When the claim was scheduled
    
    -- Verification code for document submission
    verification_code VARCHAR(20) UNIQUE, -- Unique code for verification
    code_generated_at TIMESTAMP, -- When code was generated
    code_used_at TIMESTAMP, -- When code was used
    documents_submitted_at TIMESTAMP, -- When documents were physically submitted
    
    -- Cancellation request fields
    cancellation_requested BOOLEAN DEFAULT FALSE, -- Whether applicant requested cancellation
    cancellation_reason TEXT, -- Reason for cancellation
    cancellation_requested_at TIMESTAMP, -- When cancellation was requested
    cancellation_status VARCHAR(20), -- 'pending', 'approved', 'rejected'
    cancellation_reviewed_by INTEGER REFERENCES users(id) ON DELETE SET NULL, -- Admin who reviewed
    cancellation_reviewed_at TIMESTAMP, -- When cancellation was reviewed
    cancellation_admin_notes TEXT, -- Admin notes on cancellation
    
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create application workflow status table
CREATE TABLE application_workflow_status (
    id SERIAL PRIMARY KEY,
    application_id INTEGER NOT NULL REFERENCES applications(id) ON DELETE CASCADE,
    workflow_step_id INTEGER NOT NULL REFERENCES program_workflow_steps(id) ON DELETE CASCADE,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(application_id, workflow_step_id) -- prevent duplicate step status per application
